Create a config file for project-specific settings:

```bash
squirrel init
```

This creates `squirrel.toml` in your current directory.

## Options

| Option | Alias | Description |
|--------|-------|-------------|
| `--force` | `-f` | Overwrite existing config file |
| `--project-name` | `-n` | Project name |

## Default config

The generated config contains all available settings with sensible defaults:

```toml
# Extra hostnames the crawl may follow (example.com also allows its subdomains).
# Example: domains = ["example.com"]. Empty means the seed URL domain only.
[project]
domains = []

[crawler]
max_pages = 100
delay_ms = 100
timeout_ms = 30000
user_agent = ""
robots_user_agent = "SquirrelScan"
follow_redirects = true
concurrency = 5
per_host_concurrency = 5
per_host_delay_ms = 50
max_backoff_ms = 300000
include = []
exclude = []
allow_query_params = []
drop_query_prefixes = [ "utm_", "gclid", "fbclid" ]
respect_robots = false
incremental = true
breadth_first = true
max_prefix_budget = 0.25
use_cache_control = true
max_staleness_seconds = 86400

[crawler.headers]

# Rule ids look like category/rule (for example security/csp). Globs work: eeat/* is a whole category.
# An id or glob that matches no rule is reported as a warning when you audit.
# List ids at https://docs.squirrelscan.com/rules. disable wins over enable.
# To waive a rule for some urls or items only, use an [[ignore]] table instead.
[rules]
enable = [ "*" ]
disable = []

[plugins]
enabled = false
items = []

[external_links]
enabled = true
cache_ttl_days = 7
timeout_ms = 10000
concurrency = 5

[output]
format = "console"

[storage]
keep_audits = 3

[cloud]
enabled = true
max_credits_per_audit = 1000
confirm_threshold = 50
batch_size = 20
technologies = true
editor_summary = true
domain_stats = false
render_concurrency = 6
publish = true
visibility = "unlisted"

[intel]
enabled = false
feed_ttl_hours = 24

[intel.providers]

[integrity.cloaking_probe]
enabled = false
max_pages = 10
recent_days = 14
query_variation = true
googlebot_user_agent = "Mozilla/5.0 (compatible; Googlebot/2.1; +http://www.google.com/bot.html)"

[integrity.soft404_confirm]
enabled = true
max_confirmations = 25
budget_ms = 60000

[rule_options]
```

## Examples

Create a new config:

```bash
squirrel init
```

Overwrite an existing config:

```bash
squirrel init --force
```
